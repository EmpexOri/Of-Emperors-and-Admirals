REPLACE_OR_CREATE:decree_encourage_manufacturing_industry = {
	texture = "gfx/interface/icons/decree/decree_encourage_manufacturing_industry.dds"
	country_trigger = {
		custom_tooltip = {
			text = tsarpoints_decree_not_upgraded_tt
			NOT = { has_variable = tsarpoints_upgraded_encourage_manufacturing_industry }
			NOT = { overlord ?= { has_variable = tsarpoints_upgraded_encourage_manufacturing_industry } }
		}
	}

	unlocking_technologies = {
		manufacturies
	}
	state_trigger = {
		NOT = { has_decree = decree_encourage_manufacturing_industry_tsarpoints }
	}
	modifier = {
		building_group_bg_manufacturing_throughput_add = 0.2
		building_group_bg_service_throughput_add = 0.2
		building_art_academy_throughput_add = 0.2
		building_group_bg_manufacturing_standard_of_living_add = -0.5
		building_group_bg_service_standard_of_living_add = -0.5
	}
	cost = 100

	ai_weight = {
		value = 0
		if = {
			limit = {
				any_scope_building = {
					is_building_group = bg_manufacturing
				}
			}
			add = 5
		}
		if = {
			limit = {
				any_scope_building = {
					is_building_group = bg_manufacturing
					count >= 3
				}
			}
			add = 10
		}
		if = {
			limit = {
				any_scope_building = {
					is_building_group = bg_manufacturing
					count >= 5
				}
			}
			add = 15
		}
		if = {
			limit = {
				scope:country ?= {
					has_variable_map = industrial_states_map
					is_key_in_variable_map = {
						name = industrial_states_map
						target = root
					}
				}
			}
			add = {
				value = 5
				subtract = "scope:country.variable_map(industrial_states_map|root)"
				multiply = 20
			}
		}
		if = {
			limit = {
				scope:country = {
					has_journal_entry = je_taming_the_north
				}
				state_region = {
					this = s:STATE_HOKKAIDO
				}
			}
			add = 50
		}
		if = {
			limit = {
				scope:country = {
					has_strategy = ai_strategy_industrial_expansion
				}
			}
			multiply = 1.5
		}
		if = {
			limit = {
				is_capital = yes
				scope:country = {
					has_journal_entry = je_tp_pool_develop_capital
				}
			}
			multiply = 1.5
		}
		if = {
			limit = {
				scope:country ?= {
					has_journal_entry = je_tp_fav_parochial
					has_variable = tp_fav_home_state
					var:tp_fav_home_state ?= root
				}
			}
			multiply = 1.5
		}
	}
}
