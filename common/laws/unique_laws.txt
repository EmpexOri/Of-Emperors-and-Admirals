law_choninbakufu = {
	group = lawgroup_distribution_of_power
	parent = law_oligarchy

	icon = "gfx/interface/icons/law_icons/bakufu.dds"

	progressiveness = -100

	unlocking_laws = {
		law_monarchy
		law_social_monarchy
	}

	is_visible = {
		c:JAP ?= this
		NOT = {
			has_law = law_type:law_autocracy
		}
	}

	on_activate = {
		scripted_effect_parties_disappearence = yes

		if = {
			limit = {
				country_has_monarchy_law = yes
				game_date > 1836.1.1
				c:JAP ?= this
				NOR = {
					is_revolutionary = yes
					is_secessionist = yes
				}
			}
			custom_tooltip = {
				text = shogun_returns_to_throne_tt
				country_generate_japanese_shogun = yes
			}
		}

		hidden_effect = {
			country_assess_daimyo_alignments = yes
		}
	}

	on_deactivate = {
		if = {
			limit = {
				country_has_monarchy_law = yes
				country_has_non_monarchy_law = no
			}
			custom_tooltip = {
				text = emperor_returns_to_throne_tt
				country_restore_japanese_emperor = yes
			}
			hidden_effect = {
				country_assess_daimyo_alignments = yes
			}
		}
	}

	modifier = {
		country_legitimacy_headofstate_add = 10
		country_legitimacy_govt_total_clout_add = 50
		country_legitimacy_govt_size_add = 1
		country_legitimacy_ideological_incoherence_mult = 0.5
		country_authority_add = 200

		# Merchant houses
		country_shopkeepers_pol_str_mult = 0.75
		country_capitalists_pol_str_mult = 1.5

		# Traditional landed elite
		country_aristocrats_pol_str_mult = 0.75

		country_officers_pol_str_mult = 0.10
		country_soldiers_pol_str_mult = 0.05

		interest_group_ig_petty_bourgeoisie_pol_str_mult = -0.75
		
		state_capitalists_investment_pool_efficiency_mult = 0.75
		state_aristocrats_investment_pool_efficiency_mult = 0.75

		state_shopkeepers_investment_pool_efficiency_mult = -0.75
		state_farmers_investment_pool_efficiency_mult = -1.0
		state_clergymen_investment_pool_efficiency_mult = -0.75
		state_bureaucrats_investment_pool_efficiency_mult = -1.0
	}

	can_impose = {
		always = no
	}

	ai_will_do = {
		always = no
	}
